# C 实现 KNN

用 C 从零实现：结构体数据、距离计算、选择排序取前k、多数投票

```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <time.h>

#define FEATURE_DIM 4

enum {
    METRIC_EUCLIDEAN = 0,
    METRIC_MANHATTAN = 1
} E_METRIC_TYPE;


typedef struct {
    double f[FEATURE_DIM];
    int label;              // 0,1,2
} Sample;

typedef struct {
    double distance;
    int label;
} Neighbor;

// 距离：欧氏（中间省略 sqrt）、可加开关支持曼哈顿
double dist(const Sample *a, const Sample *b, int metric);

// 预测：暴力扫描 + 选择排序取前k + 多数投票
int knn_predict(const Sample *train, int n, const Sample *x, int k, int metric);

// 主流程：打乱数据集 → 留出法划分 train/test → 算准确率
int main(void);


double dist(const Sample *a,
            const Sample *b,
            int metric)
{
    double sum = 0.0;

    for (int i = 0; i < 4; i++)
    {
        double diff = fabs(a->f[i] - b->f[i]);

        if (metric == METRIC_MANHATTAN)
        {
            sum += diff;
        }
        else
        {
            sum += diff * diff;
        }
    }

    return sum;
}

static void swap_neighbor(Neighbor *a, Neighbor *b)
{
    Neighbor tmp = *a;
    *a = *b;
    *b = tmp;
}

static void swap_sample(Sample *a, Sample *b)
{
    Sample tmp = *a;
    *a = *b;
    *b = tmp;
}

int knn_predict(const Sample *train,
                int n,
                const Sample *x,
                int k,
                int metric)
{
    Neighbor *nb = malloc(sizeof(Neighbor) * n);

    if (nb == NULL)
        return -1;

    for (int i = 0; i < n; i++) {
        nb[i].distance = dist(&train[i], x, metric);

        nb[i].label = train[i].label;
    }

    /* 只排序前k个位置 */
    for (int i = 0; i < k; i++) {
        int min_idx = i;

        for (int j = i + 1; j < n; j++) {
            if (nb[j].distance < nb[min_idx].distance) {
                min_idx = j;
            }
        }

        swap_neighbor(&nb[i], &nb[min_idx]);
    }

    int vote[3] = {0};

    for (int i = 0; i < k; i++) {
        vote[nb[i].label]++;
    }

    free(nb);

    int winner = 0;

    for (int i = 1; i < 3; i++) {
        if (vote[i] > vote[winner]) {
            winner = i;
        }
    }

    return winner;
}

//Fisher-Yates 洗牌
void shuffle(Sample *data, int n)
{
    for (int i = n - 1; i > 0; i--) {
        int j = rand() % (i + 1);

        swap_sample(&data[i], &data[j]);
    }
}

// 留出法：Fisher-Yates 洗牌 → 前70%做训练，后30%测试
double evaluate(Sample *data,
                int total,
                int k,
                int metric)
{
    shuffle(data, total);

    int train_size = (int)(total * 0.7);

    int test_size = total - train_size;

    int correct = 0;

    for (int i = train_size; i < total; i++) {
        int pred = knn_predict(data, train_size, &data[i], k, metric);

        if (pred == data[i].label) {
            correct++;
        }
    }

    return (double)correct / test_size;
}

int main(void)
{
    srand((unsigned)time(NULL));

    Sample dataset[] =
    {
        {{1,1,1,1}, 0},
        {{2,1,1,2}, 0},
        {{2,2,1,1}, 0},

        {{8,8,9,8}, 1},
        {{9,9,8,8}, 1},
        {{8,9,8,9}, 1},

        {{20,19,20,21}, 2},
        {{19,20,21,20}, 2},
        {{21,20,20,19}, 2}
    };

    int total = sizeof(dataset) / sizeof(dataset[0]);

    int k = 3;

    double acc = evaluate(dataset, total, k, METRIC_EUCLIDEAN);

    printf("Accuracy = %.2f%%\n", acc * 100.0);

    return 0;
}
```
