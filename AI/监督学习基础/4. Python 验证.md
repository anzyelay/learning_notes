```python
import math
from collections import Counter

dataset = [
    ([1,1,1,1], 0),
    ([2,1,1,2], 0),
    ([2,2,1,1], 0),

    ([8,8,9,8], 1),
    ([9,9,8,8], 1),
    ([8,9,8,9], 1),

    ([20,19,20,21], 2),
    ([19,20,21,20], 2),
    ([21,20,20,19], 2),
]

def dist(a, b):
    # 欧氏距离平方（不需要sqrt）
    return sum((x-y)**2 for x, y in zip(a, b))

def knn_predict(train, x, k=3):
    neighbors = []

    for feat, label in train:
        neighbors.append((dist(feat, x), label))

    neighbors.sort()

    votes = Counter(label for _, label in neighbors[:k])

    return votes.most_common(1)[0][0]


# 留一个样本测试
test_x, real_label = dataset[0]

train = dataset[1:]

pred = knn_predict(train, test_x, k=3)

print("真实:", real_label)
print("预测:", pred)
```

```python
import random
import numpy as np

from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score

X = np.array([
    [1,1,1,1],
    [2,1,1,2],
    [2,2,1,1],

    [8,8,9,8],
    [9,9,8,8],
    [8,9,8,9],

    [20,19,20,21],
    [19,20,21,20],
    [21,20,20,19],
])

y = np.array([
    0,0,0,
    1,1,1,
    2,2,2
])

random.seed(123)

idx = list(range(len(X)))
random.shuffle(idx)

X = X[idx]
y = y[idx]

train_size = int(len(X) * 0.7)

X_train = X[:train_size]
y_train = y[:train_size]

X_test = X[train_size:]
y_test = y[train_size:]

print("train =", len(X_train))
print("test  =", len(X_test))

knn = KNeighborsClassifier(
    n_neighbors=3,
    metric='euclidean'
)

knn.fit(X_train, y_train)

pred = knn.predict(X_test)

acc = accuracy_score(y_test, pred)

print("sklearn accuracy:",
      f"{acc*100:.2f}%")
```
