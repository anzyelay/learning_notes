# Python 实现决策树

用 python 从零实现 CART：特征选择、递归建树、预测（重点：树结构递归）

```python
# 年龄 收入 标签
# 0 = 不买
# 1 = 买
data = [
    [20, 3000, 0],
    [22, 3500, 0],
    [35, 8000, 1],
    [40, 12000, 1],
    [44, 1000, 0],
    [54, 1300, 1],
    [38, 9000, 1]
]

def gini(labels):
    total = len(labels)

    if total == 0:
        return 0

    count = {}

    for label in labels:
        count[label] = count.get(label, 0) + 1

    impurity = 1

    for c in count.values():
        p = c / total
        impurity -= p*p

    return impurity

def split(data, feature_idx, threshold):

    left = []
    right = []

    for row in data:
        if row[feature_idx] <= threshold:
            left.append(row)
        else:
            right.append(row)

    return left, right

def split_gini(left, right):

    left_labels = [ row[-1] for row in left ]
    left_gini = gini(left_labels)

    right_labels = [ row[-1] for row in right ]
    right_gini = gini(right_labels)

    total = len(left) + len(right)

    score = len(left) / total * left_gini + len(right) / total * right_gini

    return score

def find_best_split(data, min_samples_leaf=5):

    best_gini = float("inf")
    best_feature = None
    best_threshold = None

    n_features = len(data[0]) - 1

    for feature in range(n_features):
        thresholds = []
        values = sorted(set(row[feature] for row in data))

        for i in range(len(values) - 1):
            threshold = (values[i]+values[i+1]) / 2
            left, right = split(data, feature, threshold)

            if len(left) < min_samples_leaf:
                continue
            if len(right) < min_samples_leaf:
                continue

            score = split_gini(left, right)

            # print(f"特征={feature}, 阈值={threshold}, Gini={score:.3f}")

            if best_gini > score:
                best_gini = score
                best_feature = feature
                best_threshold = threshold

    return best_feature,best_threshold

class Node:

    def __init__(
            self,
            feature=None,
            threshold=None,
            left=None,
            right=None,
            value=None
    ):
        self.feature = feature
        self.threshold = threshold
        self.left = left
        self.right = right
        self.value = value

def majority_vote(labels):

    count = {}

    max_cnt = 0
    for label in labels:
        count[label] = count.get(label, 0) + 1
        if count[label] > max_cnt:
            max_cnt = count[label]
            max_label = label

    return max_label


def build_tree(data, depth=0, max_depth=5, min_samples_split=5):

    labels = [ row[-1] for row in data ]

    if len(set(labels)) == 1:
        return Node(value=labels[0])

    if depth >= max_depth:
        return Node(value=majority_vote(labels))

    if len(data) < min_samples_split:
        return Node(value=majority_vote(labels))

    feature, threshold = find_best_split(data, 3)
    if feature is None or threshold is None:
        return Node(value=majority_vote(labels))

    left, right = split(data, feature, threshold)

    right_node = build_tree(right, depth+1, max_depth, min_samples_split)
    left_node = build_tree(left, depth+1, max_depth, min_samples_split)

    return Node(feature, threshold, left_node, right_node)

def print_tree(node, indent="", feature_names=[], labels=[]):

    if node.value is not None:
        print(indent + f"{labels[node.value]}")
        return

    feature_name = feature_names[node.feature]

    print(
        indent +
        f"{feature_name} <= {node.threshold:.2f}"
    )

    print(indent + "├─ 是")
    print_tree(node.left, indent + "│  ", feature_names, labels)

    print(indent + "└─ 否")
    print_tree(node.right, indent + "   ", feature_names, labels)


root = build_tree(data)
feature_names = ["年龄", "收入"]
labels = ["不买", "买"]
print_tree(root, "", feature_names, labels)
```
